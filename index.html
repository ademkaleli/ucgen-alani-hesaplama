def ucgen_alani_hesapla():
    try:
        taban = float(input("Üçgenin taban uzunluğunu giriniz: "))
        yukseklik = float(input("Üçgenin yüksekliğini giriniz: "))

        if taban <= 0 or yukseklik <= 0:
            print("[HATA] Taban ve yükseklik 0 veya negatif olamaz!")
            return

        alan = (taban * yukseklik) / 2
        print(f"Hesaplanan Alan: {alan:.2f}")

    except ValueError:
        print("[HATA] Lütfen geçerli bir sayısal değer giriniz!")

if __name__ == "__main__":
    ucgen_alani_hesapla()
